cmake_minimum_required(VERSION 3.20)
project(multithreaded_storage_engine LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

add_library(storage src/file.cpp src/wal.cpp src/table.cpp src/engine.cpp)
target_include_directories(storage PUBLIC src)
target_link_libraries(storage PUBLIC Threads::Threads)
if(MSVC)
    target_compile_options(storage PRIVATE /W4)
else()
    target_compile_options(storage PRIVATE -Wall -Wextra -Wpedantic)
endif()

option(ENABLE_TSAN "ThreadSanitizer, Linux GCC/Clang" OFF)
if(ENABLE_TSAN)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "Za TSAN koristi Linux GCC/Clang")
    endif()
    target_compile_options(storage PUBLIC -fsanitize=thread -g -fno-omit-frame-pointer)
    target_link_options(storage PUBLIC -fsanitize=thread)
endif()

add_executable(storage_cli src/main.cpp)
target_link_libraries(storage_cli PRIVATE storage)
add_executable(storage_bench src/benchmark.cpp src/thread_pool.cpp)
target_link_libraries(storage_bench PRIVATE storage)

include(CTest)
if(BUILD_TESTING)
    add_executable(storage_tests tests/tests.cpp)
    target_link_libraries(storage_tests PRIVATE storage)
    add_test(NAME unit COMMAND storage_tests)
    set_tests_properties(unit PROPERTIES TIMEOUT 30)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME recovery COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/recovery.py" --cli $<TARGET_FILE:storage_cli>)
        set_tests_properties(recovery PROPERTIES TIMEOUT 45)
    else()
        message(WARNING "Python3 nije pronadjen; recovery testove pokreni rucno")
    endif()
endif()
